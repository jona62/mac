# Mac LSP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a Language Server Protocol server to the mac-lang VS Code extension providing diagnostics, go-to-definition, hover, and autocomplete for `.mac` files.

**Architecture:** TypeScript LSP server inside `mac-lang/`, using `vscode-languageserver`. The server re-implements the Mac scanner and parser in TypeScript for analysis. An analyzer walks the AST to build a symbol table with scope chains. The existing extension becomes the LSP client.

**Tech Stack:** TypeScript, vscode-languageserver, vscode-languageclient, npm

### Task 2: Scanner (TypeScript Port)

**Files:**
- Create: `mac-lang/src/scanner.ts`

- [ ] **Step 1: Write scanner.ts**

Port the C++ scanner. The TypeScript scanner should:
- Define `TokenType` enum with all 46 token types (matching `include/Token.h`)
- Define `Token` interface: `{ type: TokenType, lexeme: string, literal: string | number | boolean | null, line: number, column: number }`
- Define `Scanner` class with `scanTokens(): Token[]` method
- Handle all single-char tokens: `( ) { } [ ] , . - + ; / * : %`
- Handle two-char tokens: `! != = == > >= < <=`
- Handle `//` comments (skip to end of line)
- Handle string literals `"..."`
- Handle number literals (integer and decimal)
- Handle identifiers and keywords (19 keywords: `and break class continue else false fun for if in nil or print return super this true var while`)

```typescript
export enum TokenType {
  // Single-character
  LEFT_PAREN, RIGHT_PAREN, LEFT_BRACE, RIGHT_BRACE,
  LEFT_BRACKET, RIGHT_BRACKET,
  COMMA, DOT, MINUS, PLUS, SEMICOLON, SLASH, STAR,
  COLON, PERCENT,
  // One or two character
  BANG, BANG_EQUAL,
  EQUAL, EQUAL_EQUAL,
  GREATER, GREATER_EQUAL,
  LESS, LESS_EQUAL,
  // Literals
  IDENTIFIER, STRING, NUMBER,
  // Keywords
  AND, BREAK, CLASS, CONTINUE, ELSE, FALSE, FUN, FOR, IF, IN, NIL, OR,
  PRINT, RETURN, SUPER, THIS, TRUE, VAR, WHILE,
  // End
  EOF
}

export interface Token {
  type: TokenType;
  lexeme: string;
  literal: string | number | boolean | null;
  line: number;
  column: number;
}

export interface ScanError {
  line: number;
  column: number;
  message: string;
}

export class Scanner {
  private source: string;
  private tokens: Token[] = [];
  private errors: ScanError[] = [];
  private start = 0;
  private current = 0;
  private line = 1;
  private column = 0;

  private static keywords: Map<string, TokenType> = new Map([
    ["and", TokenType.AND], ["break", TokenType.BREAK],
    ["class", TokenType.CLASS], ["continue", TokenType.CONTINUE],
    ["else", TokenType.ELSE], ["false", TokenType.FALSE],
    ["fun", TokenType.FUN], ["for", TokenType.FOR],
    ["if", TokenType.IF], ["in", TokenType.IN],
    ["nil", TokenType.NIL], ["or", TokenType.OR],
    ["print", TokenType.PRINT], ["return", TokenType.RETURN],
    ["super", TokenType.SUPER], ["this", TokenType.THIS],
    ["true", TokenType.TRUE], ["var", TokenType.VAR],
    ["while", TokenType.WHILE],
  ]);

  constructor(source: string) {
    this.source = source;
  }

  scanTokens(): { tokens: Token[]; errors: ScanError[] } {
    while (!this.isAtEnd()) {
      this.start = this.current;
      this.scanToken();
    }
    this.tokens.push({
      type: TokenType.EOF, lexeme: "", literal: null,
      line: this.line, column: this.column
    });
    return { tokens: this.tokens, errors: this.errors };
  }

  // ... (port advance, match, peek, peekNext, isAtEnd, isDigit, isAlpha,
  //      isAlphaNumeric, scanToken, string, number, identifier methods
  //      from src/Scanner.cpp — same logic, TypeScript syntax)
}
```

Full implementation mirrors `src/Scanner.cpp` `scanToken()` method: single-char switch, two-char switch, `/` comment handling, `"` string handling, digit handling, identifier/keyword handling. Track column positions (C++ version only tracks line).

- [ ] **Step 2: Implement Parser class**

```typescript
export class Parser {
  private tokens: Token[];
  private current = 0;
  private errors: ParseError[] = [];

  constructor(tokens: Token[]) {
    this.tokens = tokens;
  }

  parse(): { statements: Stmt[]; errors: ParseError[] } {
    const statements: Stmt[] = [];
    while (!this.isAtEnd()) {
      const decl = this.declaration();
      if (decl) statements.push(decl);
    }
    return { statements, errors: this.errors };
  }

  // Port all parsing methods from src/Parser.cpp:
  // declaration, varDeclaration, functionDeclaration, classDeclaration,
  // statement, printStatement, ifStatement, whileStatement, forStatement,
  // expressionStatement, block,
  // primary, arrayLiteral, mapLiteral, finishCall, call, unary, factor,
  // term, comparison, equality, logicalOr, logicalAnd, assignment, expression
  //
  // Key difference from C++: no templates, return AST objects directly
  // instead of shared_ptr<Expr<T>>. Error recovery uses synchronize().
  // Lambda detection: 2-token lookahead in declaration() — only treat
  // FUN as function declaration if followed by IDENTIFIER.
}
```

Same grammar rules and precedence as C++ parser. Catch ParseError, call `synchronize()`, push error to `this.errors`, return null from declaration.

```typescript
export interface Symbol {
  name: string;
  kind: "variable" | "function" | "class" | "parameter" | "native" | "method";
  token: Token;             // declaration location
  params?: Token[];         // for functions/methods
  superclass?: string;      // for classes
  methods?: string[];       // for classes
  description?: string;     // for native functions
}

export interface Scope {
  symbols: Map<string, Symbol>;
  parent: Scope | null;
}

export interface Diagnostic {
  line: number;
  column: number;
  endColumn: number;
  message: string;
  severity: "error" | "warning";
}

export interface SymbolReference {
  token: Token;
  definition: Symbol | null;
}

export interface AnalysisResult {
  diagnostics: Diagnostic[];
  symbols: Symbol[];           // all declared symbols
  references: SymbolReference[]; // all identifier usages -> definitions
  scopes: Scope[];             // for autocomplete context
}
```

- [ ] **Step 2: Implement Analyzer class**

```typescript
export class Analyzer {
  private diagnostics: Diagnostic[] = [];
  private symbols: Symbol[] = [];
  private references: SymbolReference[] = [];
  private scopes: Scope[] = [];
  private currentScope: Scope;

  constructor() {
    // Global scope with native functions
    this.currentScope = { symbols: new Map(), parent: null };
    this.registerNatives();
  }

  analyze(statements: Stmt[]): AnalysisResult {
    for (const stmt of statements) {
      this.analyzeStmt(stmt);
    }
    return {
      diagnostics: this.diagnostics,
      symbols: this.symbols,
      references: this.references,
      scopes: this.scopes,
    };
  }

  private registerNatives() {
    const natives: [string, number, string][] = [
      ["clock", 0, "Returns current Unix time in seconds"],
      ["len", 1, "Returns length of string or array"],
      ["substr", 3, "substr(str, start, len) — extract substring"],
      ["split", 2, "split(str, delim) — split string into array"],
      ["type", 1, "type(val) — returns type name as string"],
      ["sqrt", 1, "sqrt(n) — square root"],
      ["abs", 1, "abs(n) — absolute value"],
      ["pow", 2, "pow(base, exp) — exponentiation"],
      ["floor", 1, "floor(n) — round down"],
      ["ceil", 1, "ceil(n) — round up"],
      ["push", 2, "push(arr, val) — append to array"],
      ["pop", 1, "pop(arr) — remove and return last element"],
      ["map", 2, "map(arr, fn) — transform array with function"],
      ["filter", 2, "filter(arr, fn) — filter array by predicate"],
      ["input", 1, "input(prompt) — read line from stdin"],
      ["meme", 3, "meme(template, top, bottom) — create meme"],
    ];
    for (const [name, arity, desc] of natives) {
      const params = Array.from({ length: arity }, (_, i) =>
        ({ type: TokenType.IDENTIFIER, lexeme: `arg${i}`, literal: null, line: 0, column: 0 })
      );
      this.defineSymbol({
        name, kind: "native",
        token: { type: TokenType.IDENTIFIER, lexeme: name, literal: null, line: 0, column: 0 },
        params, description: desc,
      });
    }
  }

  // Walk methods:
  // analyzeStmt(stmt) — switch on stmt.type, handle var/function/class declarations
  //   (define symbols), resolve expressions, push/pop scopes for blocks
  // analyzeExpr(expr) — switch on expr.type, resolve variable references
  //   (lookup in scope chain, add to references, report if undeclared),
  //   recurse into sub-expressions
  // beginScope() / endScope() — push/pop scope
  // defineSymbol(symbol) — add to current scope and symbols list
  // resolveVariable(token) — walk scope chain, return Symbol or null
}
```

The analyzer walks each statement:
- `var`: define symbol, analyze initializer
- `function`: define symbol (with params), begin scope, define params, analyze body, end scope
- `class`: define symbol, begin scope for methods, define `this`, analyze each method
- `block`: begin scope, analyze statements, end scope
- `forIn`: analyze iterable, begin scope, define loop var, analyze body, end scope
- Expression statements: analyze the expression

For expressions:
- `variable`: resolve in scope chain, push to `references`. If not found, add diagnostic.
- `assign`: resolve variable, analyze value
- `call`: analyze callee and args
- `get`/`set`: analyze object (no type-level property validation — Mac is dynamic)
- `lambda`: begin scope, define params, analyze body, end scope

### Task 5: LSP Server

**Files:**
- Create: `mac-lang/src/server.ts`

- [ ] **Step 1: Implement server.ts**

```typescript
import {
  createConnection, TextDocuments, ProposedFeatures,
  InitializeParams, InitializeResult, TextDocumentSyncKind,
  CompletionItem, CompletionItemKind, Hover, Definition,
  Diagnostic as LspDiagnostic, DiagnosticSeverity, Position,
} from "vscode-languageserver/node";
import { TextDocument } from "vscode-languageserver-textdocument";
import { Scanner } from "./scanner";
import { Parser } from "./parser";
import { Analyzer, AnalysisResult, Symbol } from "./analyzer";

const connection = createConnection(ProposedFeatures.all);
const documents = new TextDocuments(TextDocument);

// Cache analysis results per document URI
const analysisCache = new Map<string, AnalysisResult>();

connection.onInitialize((_params: InitializeParams): InitializeResult => {
  return {
    capabilities: {
      textDocumentSync: TextDocumentSyncKind.Full,
      completionProvider: { triggerCharacters: ["."] },
      hoverProvider: true,
      definitionProvider: true,
    },
  };
});

// Re-analyze on document change
documents.onDidChangeContent((change) => {
  const doc = change.document;
  const result = analyzeDocument(doc.getText());
  analysisCache.set(doc.uri, result);

  // Send diagnostics
  const diagnostics: LspDiagnostic[] = result.diagnostics.map((d) => ({
    range: {
      start: { line: d.line - 1, character: d.column },
      end: { line: d.line - 1, character: d.endColumn },
    },
    message: d.message,
    severity: d.severity === "error"
      ? DiagnosticSeverity.Error
      : DiagnosticSeverity.Warning,
    source: "mac",
  }));
  connection.sendDiagnostics({ uri: doc.uri, diagnostics });
});

function analyzeDocument(source: string): AnalysisResult {
  const { tokens, errors: scanErrors } = new Scanner(source).scanTokens();
  const { statements, errors: parseErrors } = new Parser(tokens).parse();
  const analyzer = new Analyzer();
  const result = analyzer.analyze(statements);

  // Add scan errors to diagnostics
  for (const err of scanErrors) {
    result.diagnostics.push({
      line: err.line, column: err.column, endColumn: err.column + 1,
      message: err.message, severity: "error",
    });
  }
  // Add parse errors to diagnostics
  for (const err of parseErrors) {
    result.diagnostics.push({
      line: err.token.line, column: err.token.column,
      endColumn: err.token.column + err.token.lexeme.length,
      message: err.message, severity: "error",
    });
  }

  return result;
}

// Go to Definition
connection.onDefinition((params) => {
  const result = analysisCache.get(params.textDocument.uri);
  if (!result) return null;

  const pos = params.position;
  const ref = result.references.find(
    (r) => r.token.line === pos.line + 1 &&
           pos.character >= r.token.column &&
           pos.character < r.token.column + r.token.lexeme.length
  );
  if (!ref?.definition || ref.definition.token.line === 0) return null;

  return {
    uri: params.textDocument.uri,
    range: {
      start: { line: ref.definition.token.line - 1, character: ref.definition.token.column },
      end: { line: ref.definition.token.line - 1, character: ref.definition.token.column + ref.definition.token.lexeme.length },
    },
  };
});

// Hover
connection.onHover((params): Hover | null => {
  const result = analysisCache.get(params.textDocument.uri);
  if (!result) return null;

  const pos = params.position;
  // Find reference at position
  const ref = result.references.find(
    (r) => r.token.line === pos.line + 1 &&
           pos.character >= r.token.column &&
           pos.character < r.token.column + r.token.lexeme.length
  );
  if (!ref?.definition) {
    // Check if hovering over a symbol definition itself
    const sym = result.symbols.find(
      (s) => s.token.line === pos.line + 1 &&
             pos.character >= s.token.column &&
             pos.character < s.token.column + s.token.lexeme.length
    );
    if (!sym) return null;
    return { contents: formatSymbol(sym) };
  }
  return { contents: formatSymbol(ref.definition) };
});

function formatSymbol(sym: Symbol): string {
  switch (sym.kind) {
    case "variable": return `**var** ${sym.name}`;
    case "parameter": return `**parameter** ${sym.name}`;
    case "function":
      return `**fun** ${sym.name}(${sym.params?.map(p => p.lexeme).join(", ") ?? ""})`;
    case "method":
      return `**method** ${sym.name}(${sym.params?.map(p => p.lexeme).join(", ") ?? ""})`;
    case "class":
      return sym.superclass
        ? `**class** ${sym.name} < ${sym.superclass}`
        : `**class** ${sym.name}`;
    case "native":
      return `**${sym.name}**(${sym.params?.map(p => p.lexeme).join(", ") ?? ""})\n\n${sym.description ?? ""}`;
    default: return sym.name;
  }
}

// Autocomplete
connection.onCompletion((params): CompletionItem[] => {
  const result = analysisCache.get(params.textDocument.uri);
  const items: CompletionItem[] = [];

  // Keywords
  const keywords = [
    "and", "break", "class", "continue", "else", "false", "fun",
    "for", "if", "in", "nil", "or", "print", "return", "super",
    "this", "true", "var", "while",
  ];
  for (const kw of keywords) {
    items.push({ label: kw, kind: CompletionItemKind.Keyword });
  }

  if (result) {
    // All known symbols
    for (const sym of result.symbols) {
      items.push({
        label: sym.name,
        kind: sym.kind === "function" || sym.kind === "native" || sym.kind === "method"
          ? CompletionItemKind.Function
          : sym.kind === "class"
          ? CompletionItemKind.Class
          : CompletionItemKind.Variable,
        detail: formatSymbol(sym),
      });
    }
  }

  return items;
});

documents.listen(connection);
connection.listen();
```

```typescript
import * as path from "path";
import { ExtensionContext } from "vscode";
import {
  LanguageClient, LanguageClientOptions,
  ServerOptions, TransportKind,
} from "vscode-languageclient/node";

let client: LanguageClient;

export function activate(context: ExtensionContext) {
  const serverModule = context.asAbsolutePath(path.join("out", "server.js"));

  const serverOptions: ServerOptions = {
    run: { module: serverModule, transport: TransportKind.ipc },
    debug: { module: serverModule, transport: TransportKind.ipc },
  };

  const clientOptions: LanguageClientOptions = {
    documentSelector: [{ scheme: "file", language: "mac" }],
  };

  client = new LanguageClient("macLanguageServer", "Mac Language Server", serverOptions, clientOptions);
  client.start();
}

export function deactivate(): Thenable<void> | undefined {
  if (!client) return undefined;
  return client.stop();
}
```
